# Filtering, ordering, limits, projection

<Tabs>
<TabItem value="rust" label="Rust">

```rust
let rows = db.query(
    Query::new("users")
        .where_filter("age", Operator::Gte, Value::Int(18))
        .order_by("name", true)
        .limit(10)
)?;

let projected = db.query_projected_zero_copy(
    Query::new("users").where_filter("age", Operator::Eq, Value::Int(30)),
    &["name".to_string(), "age".to_string()]
)?;
```

</TabItem>

```ts
const rows = await db
```
