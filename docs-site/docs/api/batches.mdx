# Atomic batch write

<Tabs>
<TabItem value="rust" label="Rust">

```rust
db.write_batch(vec![
    BatchMutation::Put { collection: "users".into(), doc_id: "u1".into(), doc: doc1 },
    BatchMutation::Delete { collection: "users".into(), doc_id: "u2".into() },
])?;
```

</TabItem>
<TabItem value="js" label="JS/TS">

```ts
const batch = db.batch();
batch
  .set(db.collection("users").doc("u1"), { name: "Alice" })
  .delete(db.collection("users").doc("u2"));
await batch.commit();
```

</TabItem>
<TabItem value="tauri" label="Tauri/React">

```ts
const batch = tauriDb.batch();
```

</TabItem>
<TabItem value="cffi" label="C-FFI">

```c

fl_batch_set(b, "users", "u1", doc1);
```

</TabItem>
